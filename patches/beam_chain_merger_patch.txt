def get_beam_by_id(
    model,
    object_id
):

    selector = (
        model.GetModelObjectSelector()
    )

    objects = (
        selector.GetAllObjectsWithType(
            ModelObject.ModelObjectEnum.BEAM
        )
    )

    while objects.MoveNext():

        beam = objects.Current

        if (
            beam.Identifier.ID
            == object_id
        ):
            return beam

    return None


def merge_beam_chain(
    model,
    member_ids
):

    members = []

    for member_id in member_ids:

        beam = get_beam_by_id(
            model,
            member_id
        )

        if beam:

            members.append(
                beam
            )

    if len(members) < 2:

        return {
            "status":
                "not_enough_members"
        }

    points = []

    for beam in members:

        points.append(
            beam.StartPoint
        )

        points.append(
            beam.EndPoint
        )

    start_point = min(
        points,
        key=lambda p: p.X
    )

    end_point = max(
        points,
        key=lambda p: p.X
    )

    merged = Beam()

    merged.StartPoint = Point(
        start_point.X,
        start_point.Y,
        start_point.Z
    )

    merged.EndPoint = Point(
        end_point.X,
        end_point.Y,
        end_point.Z
    )

    merged.Profile.ProfileString = (
        members[0].Profile.ProfileString
    )

    merged.Material.MaterialString = (
        members[0].Material.MaterialString
    )

    merged.Class = (
        members[0].Class
    )

    merged.Name = (
        members[0].Name
    )

    merged.Position.Plane = (
        members[0].Position.Plane
    )

    merged.Position.Rotation = (
        members[0].Position.Rotation
    )

    merged.Position.Depth = (
        members[0].Position.Depth
    )

    if not merged.Insert():

        return {
            "status":
                "insert_failed"
        }

    model.CommitChanges()

    for beam in members:

        beam.Delete()

    model.CommitChanges()

    return {
        "status":
            "merged",

        "new_id":
            merged.Identifier.ID,

        "deleted":
            len(members)
    }


def merge_all_safe_beam_chains(
    model,
    beam_chains
):

    merged_count = 0

    for chain in beam_chains:

        result = merge_beam_chain(
            model,
            chain["members"]
        )

        if (
            result.get(
                "status"
            )
            ==
            "merged"
        ):

            merged_count += 1

    return {
        "status":
            "merged_all_beams",

        "chains":
            merged_count
    }
